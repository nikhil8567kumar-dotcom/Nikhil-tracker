function todayKey() {
  const d = new Date();
  return `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,"0")}-${String(d.getDate()).padStart(2,"0")}`;
}
function getRank(exp) {
  for (let i = RANKS.length-1; i >= 0; i--) if (exp >= RANKS[i].min) return RANKS[i];
  return RANKS[0];
}
function getDailyQuest() {
  const seed = todayKey().split("-").reduce((a,b) => a+parseInt(b), 0);
  return DAILY_QUESTS[seed % DAILY_QUESTS.length];
}
function formatDate() {
  const n = new Date();
  const days=["SUN","MON","TUE","WED","THU","FRI","SAT"];
  const months=["JAN","FEB","MAR","APR","MAY","JUN","JUL","AUG","SEP","OCT","NOV","DEC"];
  return `${days[n.getDay()]} ${n.getDate()} ${months[n.getMonth()]} ${n.getFullYear()}`;
}

            <div style={{ fontSize:13, fontWeight:700, color:s.today.dailyQuestDone?"rgba(240,240,240,0.3)":"#efefef", lineHeight:1.4, marginBottom:12, textDecoration:s.today.dailyQuestDone?"line-through":"none" }}>{quest.task}</div>
            {s.today.dailyQuestDone
              ? <div style={{ fontFamily:"monospace", fontSize:10, color:"#39ff14", letterSpacing:1, textAlign:"center" }}>✓ COMPLETED — Well done Hunter</div>
              : <button onClick={completeDailyQuest} style={{ ...C.btn("#ffd600"), marginTop:0 }}>⚡ Mark as Complete</button>
            }
          </div>
          {/* Custom todos */}
          <div style={C.seclbl}>Your Quests (+50 EXP each)</div>
          <div style={{ display:"flex", gap:8, marginBottom:12 }}>
            <input style={C.input} value={todoInput} onChange={e=>setTodoInput(e.target.value)} onKeyDown={e=>e.key==="Enter"&&addTodo()} placeholder="Add a quest for today..." maxLength={80} />
            <button style={C.addbtn} onClick={addTodo}>+</button>
          </div>
          <div style={{ display:"flex", justifyContent:"space-between", marginBottom:10, alignItems:"center" }}>
            <span style={C.mono10}>{(s.today.todos||[]).filter(t=>t.done).length}/{(s.today.todos||[]).length} done</span>
            <button onClick={()=>setS(p=>({...p,today:{...p.today,todos:(p.today.todos||[]).filter(t=>!t.done)}}))} style={{ fontFamily:"monospace", fontSize:9, letterSpacing:1, color:"#555", background:"none", border:"1px solid #252530", borderRadius:6, padding:"4px 10px", cursor:"pointer" }}>Clear done</button>
          </div>
          {(s.today.todos||[]).length===0
            ? <div style={{ fontFamily:"monospace", fontSize:11, color:"#555", textAlign:"center", padding:"20px 0" }}>No quests yet. Add something to crush! 🔥</div>
            : [...(s.today.todos||[])].sort((a,b)=>a.done-b.done).map(t=>(
              <div key={t.id} style={{ display:"flex", alignItems:"center", gap:12, background:t.done?"rgba(57,255,20,0.04)":"#16161e", border:t.done?"1px solid rgba(57,255,20,0.2)":"1px solid #252530", borderRadius:12, padding:"13px 14px", marginBottom:8 }}>
                <div style={C.chk(t.done)} onClick={()=>toggleTodo(t.id)}>{t.done?"✓":""}</div>
                <div style={{ flex:1, fontSize:13, fontWeight:600, color:t.done?"rgba(240,240,240,0.3)":"#efefef", textDecoration:t.done?"line-through":"none", cursor:"pointer" }} onClick={()=>toggleTodo(t.id)}>{t.text}</div>
                <span style={{ fontFamily:"monospace", fontSize:9, color:t.done?"#39ff14":"#555", marginRight:4 }}>+50</span>
                <button onClick={()=>deleteTodo(t.id)} style={{ background:"none", border:"none", color:"#555", fontSize:16, cursor:"pointer" }}>✕</button>
              </div>
            ))
          }
        </div>
      )}

      {/* ── RANKS ── */}
      {tab==="ranks" && (
        <div>
          <div style={C.seclbl}>All Ranks</div>
          {RANKS.map((r,i)=>{
            const isCurrent=r.rank===rank.rank;
            const isUnlocked=s.totalExp>=r.min;
            return (
              <div key={r.rank} style={{ background:isCurrent?r.bg:"#111118", border:isCurrent?`1px solid ${r.color}50`:"1px solid #252530", borderRadius:12, padding:"14px 16px", marginBottom:8, opacity:isUnlocked?1:0.4 }}>
                <div style={{ display:"flex", alignItems:"center", justifyContent:"space-between" }}>
                  <div style={{ display:"flex", alignItems:"center", gap:12 }}>
                    <div style={{ fontSize:32, color:r.color, textShadow:isCurrent?`0 0 20px ${r.color}`:"none", fontWeight:900, width:50 }}>{r.rank}</div>
                    <div>
                      <div style={{ fontSize:14, fontWeight:700, color:isCurrent?r.color:"#efefef" }}>{r.name}</div>
                      <div style={{ fontFamily:"monospace", fontSize:9, color:"#555", letterSpacing:1, marginTop:2 }}>{r.min.toLocaleString()} — {r.max.toLocaleString()} EXP</div>
                    </div>
                  </div>
                  {isCurrent && <div style={{ fontFamily:"monospace", fontSize:9, color:r.color, background:`${r.color}15`, border:`1px solid ${r.color}30`, borderRadius:99, padding:"3px 10px" }}>CURRENT</div>}
                  {isUnlocked && !isCurrent && <div style={{ fontFamily:"monospace", fontSize:9, color:"#39ff14" }}>✓</div>}
                </div>
                {isCurrent && (
                  <div style={{ marginTop:10 }}>
                    <div style={C.barBg}><div style={C.bar(pct, r.color)} /></div>
                    <div style={{ fontFamily:"monospace", fontSize:9, color:"#555", marginTop:4, textAlign:"right" }}>{expInRank.toLocaleString()} / {expNeeded.toLocaleString()} EXP</div>
                  </div>
                )}
              </div>
            );
          })}
          <div style={{ background:"#111118", border:"1px solid #252530", borderRadius:12, padding:"14px 16px", marginTop:4 }}>
            <div style={C.seclbl}>EXP Guide</div>
            {[["Habit tick","#39ff14","+100"],["Perfect day","#39ff14","+200 bonus"],["System quest","#ffd600","up to +150"],["Custom quest","#39ff14","+50"],["PMO clean day","#39ff14","+100"],["Punishment cleared","#39ff14","+300"],["Habit untick","#ff3c3c","-100"],["Miss habits","#ff3c3c","-100 per missed"]].map(([l,c,v])=>(
              <div key={l} style={{ display:"flex", justifyContent:"space-between", padding:"6px 0", borderBottom:"1px solid #1a1a22" }}>
                <span style={{ fontFamily:"monospace", fontSize:10, color:"#555" }}>{l}</span>
                <span style={{ fontFamily:"monospace", fontSize:10, color:c, fontWeight:700 }}>{v}</span>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* ── STRUGGLES ── */}
      {tab==="struggles" && (
        <div>
          <div style={C.seclbl}>Habit Completion Rate</div>
          <div style={{ fontFamily:"monospace", fontSize:9, color:"#555", letterSpacing:1, marginBottom:14 }}>Based on saved history • Worst first</div>
          {s.history.length===0
            ? <div style={{ fontFamily:"monospace", fontSize:11, color:"#555", textAlign:"center", padding:"30px 0" }}>Save at least 1 day to see your struggles.</div>
            : (() => {
                const rates=HABITS.map(h=>{
                  const done=s.history.filter(d=>!!d.checked?.[h.id]).length;
                  return { ...h, rate:Math.round((done/s.history.length)*100) };
                }).sort((a,b)=>a.rate-b.rate);
                return rates.map(h=>{
                  const c=h.rate>=70?"#39ff14":h.rate>=40?"#ffd600":"#ff3c3c";
                  const name=h.text.replace(/^.\s/,"");
                  return (
                    <div key={h.id} style={{ display:"flex", alignItems:"center", gap:10, marginBottom:14 }}>
                      <div style={{ fontSize:12, fontWeight:700, color:"#efefef", width:150, flexShrink:0, overflow:"hidden", textOverflow:"ellipsis", whiteSpace:"nowrap" }}>{name}</div>
                      <div style={{ flex:1, height:5, background:"#252530", borderRadius:99, overflow:"hidden" }}>
                        <div style={{ height:"100%", borderRadius:99, background:c, width:`${h.rate}%`, boxShadow:`0 0 6px ${c}80` }} />
                      </div>
                      <div style={{ fontFamily:"monospace", fontSize:10, color:"#555", width:32, textAlign:"right" }}>{h.rate}%</div>
                    </div>
                  );
                });
              })()
          }
          {s.history.length>0 && (
            <div style={{ background:"#111118", border:"1px solid #252530", borderRadius:12, padding:"14px 16px", marginTop:6 }}>
              <div style={{ fontFamily:"monospace", fontSize:9, color:"#555", letterSpacing:"2px", textTransform:"uppercase", marginBottom:10 }}>Overall Stats</div>
              {[["Days saved", s.history.length],["Total penalties", s.penalties],["Best PMO streak", s.pmo.best+" days"],["Total EXP", s.totalExp.toLocaleString()]].map(([l,v])=>(
                <div key={l} style={{ display:"flex", justifyContent:"space-between", padding:"6px 0", borderBottom:"1px solid #1a1a22" }}>
                  <span style={{ fontFamily:"monospace", fontSize:10, color:"#555" }}>{l}</span>
                  <span style={{ fontFamily:"monospace", fontSize:10, color:"#39ff14", fontWeight:700 }}>{v}</span>
                </div>
              ))}
            </div>
          )}
        </div>
      )}

      {/* ── HISTORY ── */}
      {tab==="history" && (
        <div>
          <div style={C.seclbl}>Past Days</div>
          {s.history.length===0
            ? <div style={{ fontFamily:"monospace", fontSize:11, color:"#555", textAlign:"center", padding:"30px 0" }}>No history yet. Complete habits and hit Save!</div>
            : s.history.map((day,i)=>{
                const done=HABITS.filter(h=>day.checked?.[h.id]).length;
                const c=done===HABITS.length?"#39ff14":done>=4?"#ffd600":"#ff3c3c";
                return (
                  <div key={i} style={{ background:"#111118", border:"1px solid #252530", borderRadius:12, padding:"14px 16px", marginBottom:10 }}>
                    <div style={{ display:"flex", justifyContent:"space-between", marginBottom:8 }}>
                      <span style={{ fontFamily:"monospace", fontSize:10, color:"#39ff14", letterSpacing:"1.5px" }}>{day.date}</span>
                      <span style={{ fontSize:18, color:c, fontWeight:900 }}>{done}/{HABITS.length}</span>
                    </div>
                    <div style={{ display:"flex", flexWrap:"wrap" }}>
                      {HABITS.map(h=>{
                        const d=!!day.checked?.[h.id];
                        const name=h.text.replace(/^.\s/,"").split(" ").slice(0,2).join(" ");
                        return <span key={h.id} style={C.pill(d)}>{name}</span>;
                      })}
                    </div>
                    {day.note && <div style={{ fontSize:11, color:"#555", fontStyle:"italic", marginTop:8, borderTop:"1px solid #252530", paddingTop:8 }}>"{day.note}"</div>}
                  </div>
                );
              })
          }
        </div>
      )}

      {/* ── PMO ── */}
      {tab==="pmo" && (
        <div>
          <div style={{ background:"#111118", border:"1px solid #252530", borderRadius:12, padding:16, marginBottom:14, textAlign:"center" }}>
            <div style={{ fontSize:64, color:"#39ff14", textShadow:"0 0 20px rgba(57,255,20,0.5)", lineHeight:1, fontWeight:900 }}>{s.pmo.days}</div>
            <div style={{ fontFamily:"monospace", fontSize:10, color:"#555", letterSpacing:"1.5px", textTransform:"uppercase", marginTop:4 }}>Days PMO Free</div>
            <div style={{ display:"flex", justifyContent:"center", gap:30, marginTop:14 }}>
              {[["Best Streak",`${s.pmo.best} days`],["Total Resets",s.pmo.resets]].map(([l,v])=>(
                <div key={l} style={{ textAlign:"center" }}>
                  <div style={{ fontSize:22, color:"#efefef", fontWeight:900 }}>{v}</div>
                  <div style={{ fontFamily:"monospace", fontSize:9, color:"#555", letterSpacing:1, textTransform:"uppercase" }}>{l}</div>
                </div>
              ))}
            </div>
            <button onClick={addPmoDay} style={{ ...C.btn("#39ff14"), marginTop:14 }}>+ Add Clean Day (+100 EXP)</button>
            <button onClick={()=>setPmoConfirm(true)} style={{ padding:"12px 16px", background:"rgba(255,60,60,0.08)", border:"1px solid rgba(255,60,60,0.2)", borderRadius:10, color:"#ff3c3c", fontSize:13, fontWeight:700, cursor:"pointer", width:"100%", marginTop:8, fontFamily:"system-ui,sans-serif" }}>⚠ I relapsed — reset streak</button>
          </div>
          <div style={{ fontFamily:"monospace", fontSize:10, color:"#555", textAlign:"center", lineHeight:1.8 }}>
            Every day you hold is a win.<br/>
            Under 5 days → relapse = <span style={C.r}>-500 EXP</span><br/>
            5+ days → relapse = only <span style={C.y}>-100 EXP</span>
          </div>
        </div>
      )}

    </div>
  );
}
